cmake_minimum_required(VERSION 3.16)
project(MechanicalWorkersPortfolio LANGUAGES CXX)
add_library(workers Source/Workers/WorkModel.cpp Source/Workers/HeadPresenter.cpp)
target_include_directories(workers PUBLIC Source)
target_compile_features(workers PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(workers PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(workers PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(worker_tests tests/worker_tests.cpp)
target_link_libraries(worker_tests PRIVATE workers)
add_executable(worker_demo examples/work_trace.cpp)
target_link_libraries(worker_demo PRIVATE workers)
enable_testing()
add_test(NAME worker_contracts COMMAND worker_tests)
